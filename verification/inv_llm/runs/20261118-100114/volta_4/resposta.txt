```c
static int inv_extra(const argus_ctx_t *c) {
    for (int i = 0; i < ARGUS_MAX_TRACKS; i++) {
        if (c->tracks[i].active && c->tracks[i].id < 1) return 0;
    }
    if (c->next_id == 1 && c->state != ARGUS_IDLE) return 0;
    if (c->tracks[0].active && c->tracks[0].start_x >= 160) return 0;
    if (c->tracks[0].active && c->tracks[0].last_x < 0) return 0;
    if (c->tracks[0].active && c->tracks[0].last_x >= 160) return 0;
    if (c->tracks[0].active && c->tracks[0].last_y < 0) return 0;
    if (c->tracks[0].active && c->tracks[0].last_y >= 120) return 0;
    if (c->tracks[0].active && c->tracks[0].age < 1) return 0;
    if (c->tracks[0].active && c->tracks[0].age > ARGUS_MAX_AGE) return 0;
    if (c->tracks[0].active && c->tracks[0].missed < 0) return 0;
    if (c->tracks[0].active && c->tracks[0].missed > ARGUS_MAX_MISSED) return 0;
    return 1;
}
```

A nova versão de `inv_extra` mantém as verificações originais, garantindo que o estado de partida inalcançável seja excluído. Ela se preserva em um frame porque verifica propriedades que devem ser verdadeiras em qualquer estado válido do sistema, como a validade dos IDs dos rastreamentos, a posição dos rastreamentos dentro dos limites do frame, e a idade e o número de frames perdidos dos rastreamentos. Essas verificações são essenciais para manter a integridade do sistema e garantir que o estado do sistema permaneça consistente após cada atualização.
